#include<iostream>
#include<algorithm>
using namespace std;
int main(){
    ios::sync_with_stdio(false);
    cin.tie(NULL);
    long int n,i;
    cin>>n;
    long int arr[3];
    for(i=0;i<n;i++){
        cin>>arr[0]>>arr[1]>>arr[2];
        sort(arr,arr+3);
        if(arr[2]==(arr[0]+arr[1]))cout<<"YES\n";
        else if((arr[0]==arr[1])&&((arr[2]%2)==0)||
                (arr[2]==arr[1])&&((arr[0]%2)==0)||
                (arr[0]==arr[2])&&((arr[1]%2)==0))cout<<"YES\n";
        else cout<<"NO\n";    
    }
}
